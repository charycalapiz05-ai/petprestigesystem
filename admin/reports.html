// Helper: Get filtered orders function getFilteredOrders() { let orders = JSON.parse(localStorage.getItem('petPrestigeOrders') || '[]'); const from = document.getElementById('dateFrom')?.value; const to = document.getElementById('dateTo')?.value; if (from) orders = orders.filter(o => new Date(o.date) >= new Date(from)); if (to) orders = orders.filter(o => new Date(o.date) <= new Date(to + 'T23:59:59')); return orders; } // Calculate summary stats function calculateStats(orders) { const totalRevenue = orders.reduce((sum, o) => sum + (o.total || 0), 0); const totalOrders = orders.length; const avgOrderValue = totalOrders ? (totalRevenue / totalOrders).toFixed(2) : 0; const customers = new Set(orders.map(o => o.shippingInfo?.email || '').filter(Boolean)); const uniqueCustomers = customers.size; return { totalRevenue, totalOrders, avgOrderValue, uniqueCustomers }; } // Render all charts and stats function updateReports() { const orders = getFilteredOrders(); const stats = calculateStats(orders); // Update cards document.getElementById('totalRevenue').textContent = '₱' + stats.totalRevenue.toLocaleString(); document.getElementById('totalOrders').textContent = stats.totalOrders; document.getElementById('avgOrderValue').textContent = '₱' + stats.avgOrderValue; document.getElementById('uniqueCustomers').textContent = stats.uniqueCustomers; renderCharts(orders); } // Chart instance registry to prevent memory leaks const reportCharts = {}; function destroyChart(id) { if (reportCharts[id]) { reportCharts[id].destroy(); delete reportCharts[id]; } } // Render charts function renderCharts(orders) { // Destroy all existing chart instances first ['salesTrendChart','statusChart','topProductsChart','categoryChart'].forEach(destroyChart); // 1. Sales Trend (monthly) const salesByMonth = {}; orders.forEach(o => { const month = new Date(o.date).toLocaleString('default', { month: 'short', year: 'numeric' }); salesByMonth[month] = (salesByMonth[month] || 0) + (o.total || 0); }); const salesTrendCtx = document.getElementById('salesTrendChart').getContext('2d'); reportCharts['salesTrendChart'] = new Chart(salesTrendCtx, { type: 'line', data: { labels: Object.keys(salesByMonth), datasets: [{ label: 'Revenue (₱)', data: Object.values(salesByMonth), borderColor: '#FF6B35', backgroundColor: 'rgba(255,107,53,0.2)', tension: 0.3, fill: true }] }, options: { responsive: true, scales: { y: { beginAtZero: true } } } }); // 2. Orders by Status (doughnut) const statusCount = { pending:0, processing:0, shipped:0, delivered:0, cancelled:0 }; orders.forEach(o => statusCount[o.status] = (statusCount[o.status] || 0) + 1); const statusCtx = document.getElementById('statusChart').getContext('2d'); reportCharts['statusChart'] = new Chart(statusCtx, { type: 'doughnut', data: { labels: ['Pending', 'Processing', 'Shipped', 'Delivered', 'Cancelled'], datasets: [{ data: Object.values(statusCount), backgroundColor: ['#f39c12', '#3498db', '#9b59b6', '#27ae60', '#e74c3c'] }] }, options: { responsive: true, plugins: { legend: { position: 'bottom' } } } }); // 3. Top Products (bar) const productSales = {}; orders.forEach(o => { o.items?.forEach(item => { productSales[item.name] = (productSales[item.name] || 0) + (item.quantity * item.price); }); }); const topProducts = Object.entries(productSales) .sort((a,b) => b[1] - a[1]) .slice(0, 8); const topCtx = document.getElementById('topProductsChart').getContext('2d'); reportCharts['topProductsChart'] = new Chart(topCtx, { type: 'bar', data: { labels: topProducts.map(([name]) => name.substring(0,20) + (name.length > 20 ? '...' : '')), datasets: [{ label: 'Revenue (₱)', data: topProducts.map(([, val]) => val), backgroundColor: '#FF6B35' }] }, options: { responsive: true, indexAxis: 'y', scales: { x: { beginAtZero: true } } } }); // 4. Category Distribution (pie) const categoryCount = {}; const products = [...DB.products, ...JSON.parse(localStorage.getItem('petPrestigeCustomProducts') || '[]')]; orders.forEach(o => { o.items?.forEach(item => { const prod = products.find(p => p.id === item.id); if (prod?.category) { categoryCount[prod.category] = (categoryCount[prod.category] || 0) + 1; } }); }); const catCtx = document.getElementById('categoryChart').getContext('2d'); reportCharts['categoryChart'] = new Chart(catCtx, { type: 'pie', data: { labels: Object.keys(categoryCount), datasets: [{ data: Object.values(categoryCount), backgroundColor: ['#FF6B35', '#4ECDC4', '#FFE66D', '#6BCB77', '#FF6B6B', '#9B59B6'] }] }, options: { responsive: true, plugins: { legend: { position: 'bottom' } } } }); } function exportReportSummary() { const orders = getFilteredOrders(); const stats = calculateStats(orders); const headers = ['Metric', 'Value']; const rows = [ ['Total Revenue', stats.totalRevenue], ['Total Orders', stats.totalOrders], ['Average Order Value', stats.avgOrderValue], ['Unique Customers', stats.uniqueCustomers] ]; const csv = [headers, ...rows].map(row => row.join(',')).join('\n'); const blob = new Blob([csv], { type: 'text/csv' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'petprestige_report_summary.csv'; a.click(); URL.revokeObjectURL(url); alert('Summary exported as CSV'); } function logoutAdmin() { if (confirm('Logout from admin panel?')) { localStorage.removeItem('petPrestigeAdminSession'); localStorage.removeItem('petPrestigeUser'); window.location.href = '../login.html?logout=1'; } } document.addEventListener('DOMContentLoaded', function() { updateReports(); });
